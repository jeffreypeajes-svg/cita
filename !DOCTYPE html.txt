<script>
    const TU_TELEFONO = "TU_NUMERO_AQUI"; 
    let noAttempts = 0;

    // Variables para el Calendario
    let currentDate = new Date();
    let selectedDate = null;

    function handleNoInteraction() {
        const btnNo = document.getElementById('btnNo');
        noAttempts++;

        if (noAttempts === 1) {
            btnNo.innerText = "¿Estás segura que eliges el no? 🤔";
        } else if (noAttempts === 2) {
            btnNo.innerText = "Piénsalo bien... 💭";
        } else {
            btnNo.innerText = "Tú no eres la que sabes 😜";
        }

        const x = Math.random() * (window.innerWidth - btnNo.clientWidth - 50);
        const y = Math.random() * (window.innerHeight - btnNo.clientHeight - 50);
        
        btnNo.style.position = 'fixed';
        btnNo.style.left = `${Math.max(20, Math.min(x, window.innerWidth - 180))}px`;
        btnNo.style.top = `${Math.max(20, Math.min(y, window.innerHeight - 80))}px`;
    }

    function goToStep(stepNumber) {
        document.querySelectorAll('.card > div').forEach(el => el.classList.add('hidden'));
        document.getElementById(`step${stepNumber}`).classList.remove('hidden');
        if (stepNumber === 3) {
            renderCalendar();
        }
    }

    function selectOption(element) {
        document.querySelectorAll('.option-box').forEach(el => el.classList.remove('active'));
        element.classList.add('active');
        element.querySelector('input').checked = true;
    }

    /* Lógica del Calendario Visual */
    function renderCalendar() {
        const monthHeader = document.getElementById('calendarMonth');
        const daysContainer = document.getElementById('calendarDays');
        daysContainer.innerHTML = '';

        const year = currentDate.getFullYear();
        const month = currentDate.getMonth();

        const monthNames = ["Enero", "Febrero", "Marzo", "Abril", "Mayo", "Junio", "Julio", "Agosto", "Septiembre", "Octubre", "Noviembre", "Diciembre"];
        monthHeader.innerText = `${monthNames[month]} ${year}`;

        const firstDay = new Date(year, month, 1).getDay();
        const totalDays = new Date(year, month + 1, 0).getDate();
        const today = new Date();
        today.setHours(0,0,0,0);

        for (let i = 0; i < firstDay; i++) {
            const emptyCell = document.createElement('div');
            emptyCell.classList.add('day-cell', 'empty');
            daysContainer.appendChild(emptyCell);
        }

        for (let day = 1; day <= totalDays; day++) {
            const dayCell = document.createElement('div');
            dayCell.classList.add('day-cell');
            dayCell.innerText = day;

            const thisDate = new Date(year, month, day);

            if (thisDate < today) {
                dayCell.classList.add('disabled');
            } else {
                if (selectedDate && thisDate.getTime() === selectedDate.getTime()) {
                    dayCell.classList.add('selected');
                }
                dayCell.onclick = () => {
                    document.querySelectorAll('.day-cell').forEach(c => c.classList.remove('selected'));
                    dayCell.classList.add('selected');
                    selectedDate = new Date(year, month, day);
                };
            }
            daysContainer.appendChild(dayCell);
        }
    }

    function changeMonth(delta) {
        currentDate.setMonth(currentDate.getMonth() + delta);
        renderCalendar();
    }

    function validateDateAndNext() {
        if (!selectedDate) {
            alert("Por favor selecciona un día del calendario");
            return;
        }
        goToStep(4);
    }

    function finishSelection() {
        const timeInput = document.getElementById('time').value;
        if (!timeInput) {
            alert("Por favor selecciona una hora");
            return;
        }

        const place = document.querySelector('input[name="place"]:checked').value;
        const time = timeInput;

        const formattedDate = selectedDate.toLocaleDateString('es-ES', {
            weekday: 'long', year: 'numeric', month: 'long', day: 'numeric'
        });

        document.getElementById('resPlace').innerText = place;
        document.getElementById('resDate').innerText = formattedDate;
        document.getElementById('resTime').innerText = time;

        goToStep(5);

        confetti({
            particleCount: 120,
            spread: 70,
            origin: { y: 0.6 },
            colors: ['#f472b6', '#ec4899', '#f43f5e', '#fbcfe8']
        });

        const mensaje = `¡Hola! Acepto la cita 💕\n\n` +
                        `📌 *Plan:* ${place}\n` +
                        `📅 *Fecha:* ${formattedDate}\n` +
                        `⏰ *Hora:* ${time}\n\n` +
                        `¡Nos vemos pronto! ✨`;

        const encodedMessage = encodeURIComponent(mensaje);
        const whatsappUrl = `https://api.whatsapp.com/send?phone=${TU_TELEFONO}&text=${encodedMessage}`;
        
        document.getElementById('whatsappBtn').href = whatsappUrl;
    }
</script>
